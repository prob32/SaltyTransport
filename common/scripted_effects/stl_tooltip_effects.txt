# ========================================
# PARTNER DISPLAY LIST (v2)
#
# Builds the trade-partner breakdown for ONE good on demand (click in the
# trade window, not hover — a full Bellman-Ford on hover caused UI lag).
#
# Fully generic: the good arrives as a flag target in scope:stl_goods_flag
# (set by the scripted_gui via AddScope + MakeScopeFlag(Goods.GetKey)), and
# all prices come from the Phase B caches (stl_price / stl_econ /
# stl_market_price maps) — no per-good code, no sg: reads.
#
# Direction comes from var:stl_display_is_import on the depot state
# (1 = import view: cheap partners; 0 = export view: expensive partners).
#
# Output, on the depot state:
#   stl_display_partners        variable list of partner states
#   stl_display_total_wv        sum of weighted values
#   stl_display_partner_count   number of partners
# Output, on each partner state:
#   stl_display_wv              weighted value (eff x proximity)
#   stl_display_friction        cumulative BF friction
#   stl_display_weight          proximity weight (100 - friction)
#   stl_display_source_price    partner's cached local price
#   stl_display_contribution_pct  share of total wv
#   stl_display_has_rail / stl_display_has_river
# ========================================

# Scope: depot state. Requires scope:stl_goods_flag.
stl_build_partner_display = {
	save_scope_as = stl_depot_state

	stl_clear_display_lists = yes
	set_variable = { name = stl_display_total_wv value = 0 }
	set_variable = { name = stl_display_partner_count value = 0 }

	# origin's network price for this good (default 0 = no data, no partners)
	set_local_variable = { name = stl_origin_mkt value = 0 }
	if = {
		limit = { is_key_in_variable_map = { name = stl_market_price key = scope:stl_goods_flag } }
		set_local_variable = { name = stl_origin_mkt value = variable_map(stl_market_price|scope:stl_goods_flag) }
	}

	if = {
		limit = { local_var:stl_origin_mkt > 0.01 }

		stl_run_bellman_ford = yes

		owner = {
			market = {
				every_scope_country = {
					every_scope_state = {
						limit = {
							has_variable = stl_trade_depot_ref
							NOT = { this = scope:stl_depot_state }
							scope:stl_depot_state = {
								is_key_in_variable_map = { name = stl_bf_dist key = prev }
							}
							is_key_in_variable_map = { name = stl_econ key = scope:stl_goods_flag }
						}

						# proximity weight
						scope:stl_depot_state = {
							set_local_variable = {
								name = stl_pw
								value = {
									value = 100
									subtract = variable_map(stl_bf_dist|prev)
								}
							}
						}
						if = {
							limit = { local_var:stl_pw > 0 }

							# partner's cached price & econ scale
							set_local_variable = { name = stl_pp value = variable_map(stl_price|scope:stl_goods_flag) }
							set_local_variable = { name = stl_pe value = variable_map(stl_econ|scope:stl_goods_flag) }

							# eff = (origin_market - partner_price) / origin_market x econ
							set_local_variable = {
								name = stl_eff
								value = {
									value = local_var:stl_origin_mkt
									subtract = local_var:stl_pp
									divide = local_var:stl_origin_mkt
									multiply = local_var:stl_pe
								}
							}

							set_local_variable = { name = stl_wv value = 0 }
							# import view: positive eff = cheap partner = supply source
							if = {
								limit = {
									scope:stl_depot_state = { var:stl_display_is_import >= 1 }
									local_var:stl_eff > 0
								}
								set_local_variable = {
									name = stl_wv
									value = { value = local_var:stl_eff multiply = local_var:stl_pw }
								}
							}
							# export view: negative eff = expensive partner = demand sink
							if = {
								limit = {
									scope:stl_depot_state = { NOT = { var:stl_display_is_import >= 1 } }
									NOT = { local_var:stl_eff >= 0 }
								}
								set_local_variable = { name = stl_neg value = { value = 0 subtract = local_var:stl_eff } }
								set_local_variable = {
									name = stl_wv
									value = { value = local_var:stl_neg multiply = local_var:stl_pw }
								}
							}

							if = {
								limit = { local_var:stl_wv > 0.01 }
								set_variable = { name = stl_display_wv value = local_var:stl_wv }
								set_variable = { name = stl_display_source_price value = local_var:stl_pp }
								scope:stl_depot_state = {
									set_local_variable = { name = stl_friction value = variable_map(stl_bf_dist|prev) }
								}
								set_variable = { name = stl_display_friction value = local_var:stl_friction }
								set_variable = { name = stl_display_weight value = local_var:stl_pw }
								set_variable = { name = stl_display_has_rail value = 0 }
								if = {
									limit = { has_building = building_railway }
									set_variable = { name = stl_display_has_rail value = 1 }
								}
								set_variable = { name = stl_display_has_river value = 0 }
								if = {
									limit = { stl_has_river = yes }
									set_variable = { name = stl_display_has_river value = 1 }
								}
								scope:stl_depot_state = {
									add_to_variable_list = { name = stl_display_partners target = prev }
									change_variable = { name = stl_display_total_wv add = prev.var:stl_display_wv }
									change_variable = { name = stl_display_partner_count add = 1 }
								}
							}
						}
					}
				}
			}
		}

		stl_cleanup_bf_dist = yes

		# second pass: contribution %
		if = {
			limit = { var:stl_display_total_wv > 0.01 }
			every_in_list = {
				variable = stl_display_partners
				set_variable = {
					name = stl_display_contribution_pct
					value = {
						value = var:stl_display_wv
						divide = scope:stl_depot_state.var:stl_display_total_wv
						multiply = 100
					}
				}
			}
		}
	}

	clear_saved_scope = stl_depot_state
}

# Remove display data from previous use.
# Scope: depot state
stl_clear_display_lists = {
	every_in_list = {
		variable = stl_display_partners
		remove_variable = stl_display_wv
		remove_variable = stl_display_source_price
		remove_variable = stl_display_friction
		remove_variable = stl_display_weight
		remove_variable = stl_display_contribution_pct
		remove_variable = stl_display_has_rail
		remove_variable = stl_display_has_river
	}
	clear_variable_list = stl_display_partners
}
