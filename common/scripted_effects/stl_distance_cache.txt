# ========================================
# BELLMAN-FORD INFRASTRUCTURE FRICTION PATHFINDING (v2)
#
# Computes shortest-path friction from one origin state to every
# reachable depot state in the same market.
#
# v2: distances live in a variable map ON THE ORIGIN, keyed by state.
#   - no 999-initialization sweep over the market
#   - no cleanup sweep over other countries' states
#   - concurrent use (tooltip refresh vs monthly cycle) cannot collide
#     across states; the origin clears only its own map
#
# Friction per state = max(1, 5 x (infra_usage / infra))
#   spare capacity ~1, fully used = 5, over-capacity > 5, none = 10
# Weight = max(0, 100 - cumulative_friction); >= 100 = unreachable.
#
# Scope: state (the BF origin). Read results from the origin:
#   is_key_in_variable_map = { name = stl_bf_dist key = <state> }
#   variable_map(stl_bf_dist|<state>)
# Call stl_cleanup_bf_dist (self-clear) when done.
# ========================================

stl_run_bellman_ford = {
	save_scope_as = stl_bf_origin
	owner = { market = { save_scope_as = stl_bf_market } }

	# Reset map; origin distance = 0
	if = {
		limit = { has_variable_map = stl_bf_dist }
		clear_variable_map = stl_bf_dist
	}
	add_to_variable_map = { name = stl_bf_dist key = scope:stl_bf_origin value = 0 }

	# ===== RELAXATION PASSES (7 passes for up to 7-hop paths) =====
	set_local_variable = { name = stl_bf_pass value = 0 }
	while = {
		limit = {
			NOT = { local_var:stl_bf_pass >= 7 }
		}
		change_local_variable = { name = stl_bf_pass add = 1 }

		# For each known state S, relax edges to all neighbors N
		scope:stl_bf_market = {
			every_scope_country = {
				every_scope_state = {
					limit = {
						has_variable = stl_trade_depot_ref
						# S is in the frontier (already reached)...
						scope:stl_bf_origin = {
							is_key_in_variable_map = { name = stl_bf_dist key = prev }
						}
					}

					# read S's distance while S is unambiguous (prev inside origin block)
					scope:stl_bf_origin = {
						set_local_variable = { name = stl_s_dist value = variable_map(stl_bf_dist|prev) }
					}

					# ...and not already past the reachability cutoff
					if = {
						limit = { NOT = { local_var:stl_s_dist >= 100 } }

						every_neighbouring_state = {
							limit = {
								has_variable = stl_trade_depot_ref
								owner = { market = { this = scope:stl_bf_market } }
							}

							# friction for entering this neighbor
							if = {
								limit = { infrastructure >= 1 }
								set_local_variable = {
									name = stl_bf_friction
									value = {
										value = infrastructure_usage
										divide = infrastructure
										multiply = 5
									}
								}
								if = {
									limit = { NOT = { local_var:stl_bf_friction >= 1 } }
									set_local_variable = { name = stl_bf_friction value = 1 }
								}
							}
							if = {
								limit = { NOT = { infrastructure >= 1 } }
								set_local_variable = { name = stl_bf_friction value = 10 }
							}

							set_local_variable = {
								name = stl_bf_new_dist
								value = {
									value = local_var:stl_s_dist
									add = local_var:stl_bf_friction
								}
							}

							# relax: set if unknown or improved (prev = N inside origin block)
							scope:stl_bf_origin = {
								if = {
									limit = {
										OR = {
											NOT = { is_key_in_variable_map = { name = stl_bf_dist key = prev } }
											variable_map(stl_bf_dist|prev) > local_var:stl_bf_new_dist
										}
									}
									add_to_variable_map = { name = stl_bf_dist key = prev value = local_var:stl_bf_new_dist }
								}
							}
						}
					}
				}
			}
		}
	}

	clear_saved_scope = stl_bf_origin
	clear_saved_scope = stl_bf_market
}

# Drop the origin's distance map. Cheap (single map on self) — call when done
# so saves don't accumulate O(states) keys per state.
# Scope: state (same state that called stl_run_bellman_ford)
stl_cleanup_bf_dist = {
	if = {
		limit = { has_variable_map = stl_bf_dist }
		clear_variable_map = stl_bf_dist
	}
}
