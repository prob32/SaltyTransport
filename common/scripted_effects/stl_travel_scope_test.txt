# ========================================
# TRAVEL SCOPE PROBE (DEBUG / DISPOSABLE)
#
# Determines whether the engine ever places travel_node /
# travel_connection (or *_definition) into script scope.
#
#   - error_log writes to error.log ALWAYS (no -debug_mode needed),
#     so the PRESENT/absent results are always visible.
#   - debug_log_scopes = yes dumps the FULL scope tree to game.log,
#     but only when the game is launched with -debug_mode.
#
# Every line is tagged STL_TRAVEL_PROBE. No '[' ']' in log strings
# (they are parsed as data promotes and error out).
# ========================================

stl_travel_scope_probe = {
	error_log = "STL_TRAVEL_PROBE begin_scope_dump"

	# Full scope tree (ROOT/PREV/all saved scopes + types) -> game.log,
	# requires -debug_mode. Harmless if debug mode is off.
	debug_log_scopes = yes

	# Direct probes of the four documented Save Tokens.
	# exists = scope:<name> safely returns false if not present.
	if = {
		limit = { exists = scope:travel_node }
		error_log = "STL_TRAVEL_PROBE travel_node_PRESENT"
	}
	else = { error_log = "STL_TRAVEL_PROBE travel_node_absent" }

	if = {
		limit = { exists = scope:travel_node_definition }
		error_log = "STL_TRAVEL_PROBE travel_node_definition_PRESENT"
	}
	else = { error_log = "STL_TRAVEL_PROBE travel_node_definition_absent" }

	if = {
		limit = { exists = scope:travel_connection }
		error_log = "STL_TRAVEL_PROBE travel_connection_PRESENT"
	}
	else = { error_log = "STL_TRAVEL_PROBE travel_connection_absent" }

	if = {
		limit = { exists = scope:travel_connection_definition }
		error_log = "STL_TRAVEL_PROBE travel_connection_definition_PRESENT"
	}
	else = { error_log = "STL_TRAVEL_PROBE travel_connection_definition_absent" }

	error_log = "STL_TRAVEL_PROBE end_scope_dump"
}
