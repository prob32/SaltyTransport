# ========================================
# SALTY TRANSPORT ON_ACTIONS
# Building creation handled by common/history/buildings/stl_global.txt
# These hooks handle ongoing init and distribution
#
# Two-phase monthly cycle:
#   Phase B (days 8-20): BF + effective surplus WAS/WAD + market price
#   Phase C (days 21-28): Normalize + apply modifiers
# ========================================

# Yearly pulse — safety net for states acquired mid-game
on_yearly_pulse_country = {
	on_actions = {
		stl_oa_yearly_init
	}
}

stl_oa_yearly_init = {
	effect = {
		# Init depots (idempotent — skips if already done)
		trigger_event = { id = stl_events.1 }
	}
}

# Monthly pulse — init + schedule three-phase distribution cycle
on_monthly_pulse = {
	on_actions = {
		stl_oa_monthly_pulse
	}
}

stl_oa_monthly_pulse = {
	effect = {
		# ALL countries: init depots (idempotent safety net)
		every_country = {
			trigger_event = { id = stl_events.1 }
		}

		# MARKET OWNERS ONLY: run three-phase distribution
		every_country = {
			limit = {
				NOT = { is_country_type = decentralized }
				this = market.owner
			}

			# Phase B: days 8-20 (state BF batches, ~8 states/day)
			trigger_event = { id = stl_events.3 days = 8 }
			trigger_event = { id = stl_events.3 days = 9 }
			trigger_event = { id = stl_events.3 days = 10 }
			trigger_event = { id = stl_events.3 days = 11 }
			trigger_event = { id = stl_events.3 days = 12 }
			trigger_event = { id = stl_events.3 days = 13 }
			trigger_event = { id = stl_events.3 days = 14 }
			trigger_event = { id = stl_events.3 days = 15 }
			trigger_event = { id = stl_events.3 days = 16 }
			trigger_event = { id = stl_events.3 days = 17 }
			trigger_event = { id = stl_events.3 days = 18 }
			trigger_event = { id = stl_events.3 days = 19 }
			trigger_event = { id = stl_events.3 days = 20 }

			# Phase C: days 21-28 (normalize + apply, ~6 goods/batch)
			trigger_event = { id = stl_events.4 days = 21 }
			trigger_event = { id = stl_events.4 days = 22 }
			trigger_event = { id = stl_events.4 days = 23 }
			trigger_event = { id = stl_events.4 days = 24 }
			trigger_event = { id = stl_events.4 days = 25 }
			trigger_event = { id = stl_events.4 days = 26 }
			trigger_event = { id = stl_events.4 days = 27 }
			trigger_event = { id = stl_events.4 days = 28 }
		}
	}
}
