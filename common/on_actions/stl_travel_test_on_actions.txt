# ========================================
# TRAVEL SCOPE PROBE - ON_ACTION HOOKS (DEBUG / DISPOSABLE)
#
# These five on_actions fire when a travel order is CANCELLED.
# They are the ONLY documented places the engine might hand a
# travel_node / travel_connection into script context.
#
# Uses the on_actions={} list pattern (matches stl_on_actions.txt)
# so we APPEND our handler instead of overwriting the base effect.
#
# REPRODUCE: order a formation to travel (deploy to front, deploy
# to state, deploy to sea node, station in HQ, or track a formation),
# then CANCEL it before it arrives.
#
# NOTE: these on_actions only exist in Victoria 3 1.13.x.
# Output goes to game.log / debug.log (NOT error.log). Grep STL_TRAVEL_PROBE.
# ========================================

on_travel_to_front_cancelled = {
	on_actions = { stl_oa_travel_to_front_cancelled }
}
stl_oa_travel_to_front_cancelled = {
	effect = {
		debug_log = "STL_TRAVEL_PROBE oa_travel_to_front_cancelled"
		stl_travel_scope_probe = yes
	}
}
