# ========================================
# TRAVEL SCOPE PROBE - ON_ACTION HOOKS (DEBUG / DISPOSABLE)
#
# These five on_actions fire when a travel order is CANCELLED -
# the only documented places the engine might hand a travel_node /
# travel_connection into script context.
#
# Uses on_actions={} list pattern so we APPEND (don't overwrite).
# error_log output -> error.log (always). debug_log_scopes -> game.log
# (only with -debug_mode).
#
# REPRODUCE: order a formation to travel (deploy to front/state/sea
# node, station in HQ, or track a formation), then CANCEL it.
# NOTE: these on_actions only exist in Victoria 3 1.13.x.
# ========================================

on_travel_to_front_cancelled = {
	on_actions = { stl_oa_travel_to_front_cancelled }
}
stl_oa_travel_to_front_cancelled = {
	effect = {
		error_log = "STL_TRAVEL_PROBE oa_travel_to_front_cancelled"
		stl_travel_scope_probe = yes
	}
}
