# Five generic entries replace the v0.1 per-good set (2 x 49 generated).
# The good is passed from the GUI as a flag scope:
#   datacontext = "[GetScriptedGui('stl_sgui_show_import_partners')]"
#   onclick = "[ScriptedGui.Execute(GuiScope.SetRoot(State.MakeScope).AddScope('stl_goods_flag', MakeScopeFlag(Goods.GetKey)).End)]"
# ========================================

# Build the partner breakdown for one good, import view (cheap partners).
stl_sgui_show_import_partners = {
	scope = state
	effect = {
		set_variable = { name = stl_display_is_import value = 1 }
		stl_build_partner_display = yes
	}
	is_shown = { always = yes }
	is_valid = { always = yes }
	ai_is_valid = { always = no }
}

# Build the partner breakdown for one good, export view (expensive partners).
stl_sgui_show_export_partners = {
	scope = state
	effect = {
		set_variable = { name = stl_display_is_import value = 0 }
		stl_build_partner_display = yes
	}
	is_shown = { always = yes }
	is_valid = { always = yes }
	ai_is_valid = { always = no }
}

# Tear down the partner breakdown (panel closed / selection changed).
stl_sgui_clear_display = {
	scope = state
	effect = {
		stl_clear_display_lists = yes
	}
	is_shown = { always = yes }
	is_valid = { always = yes }
	ai_is_valid = { always = no }
}

# Open the trade window for a state. Executed from the depot building panel:
#   GuiScope.SetRoot(GetPlayer.MakeScope).AddScope('stl_target_state', Building.GetState.MakeScope).End
stl_sgui_open_window = {
	scope = country
	effect = {
		set_variable = { name = stl_window_state value = scope:stl_target_state }
	}
	is_shown = { always = yes }
	is_valid = { always = yes }
	ai_is_valid = { always = no }
}

# Close the trade window.
stl_sgui_close_window = {
	scope = country
	is_valid = {
		has_variable = stl_window_state
	}
	effect = {
		remove_variable = stl_window_state
	}
	ai_is_valid = { always = no }
}
